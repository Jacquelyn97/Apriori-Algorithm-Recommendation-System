-- 创建数据库
CREATE DATABASE IF NOT EXISTS tea_shop
  DEFAULT CHARSET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE tea_shop;

-- 1. 用户与画像
CREATE TABLE users (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  openid        VARCHAR(64) NOT NULL UNIQUE,
  nickname      VARCHAR(100),
  avatar_url    VARCHAR(255),
  channel       VARCHAR(50),              -- miniapp / web 等
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE user_profile (
  user_id        BIGINT UNSIGNED NOT NULL PRIMARY KEY,
  gender         ENUM('M','F','U') DEFAULT 'U',
  age            INT,
  occupation     VARCHAR(100),
  city           VARCHAR(100),
  province       VARCHAR(100),
  income_level   VARCHAR(50),
  marital_status VARCHAR(50),
  created_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_user_profile_user
    FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. 门店
CREATE TABLE stores (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  code        VARCHAR(50) NOT NULL UNIQUE,   -- kuchai / sri_petal / cheras
  name        VARCHAR(100) NOT NULL,
  address     VARCHAR(255),
  city        VARCHAR(100),
  lat         DECIMAL(10,7),
  lng         DECIMAL(10,7),
  open_hours  VARCHAR(100),
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. 商品类目与商品
CREATE TABLE product_categories (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name          VARCHAR(100) NOT NULL,
  parent_id     BIGINT UNSIGNED,
  display_order INT DEFAULT 0,
  CONSTRAINT fk_prod_cat_parent
    FOREIGN KEY (parent_id) REFERENCES product_categories(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE products (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  category_id   BIGINT UNSIGNED,
  name          VARCHAR(150) NOT NULL,
  base_price    DECIMAL(10,2) NOT NULL,
  image_url     VARCHAR(255),
  is_active     TINYINT(1) NOT NULL DEFAULT 1,
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_product_category
    FOREIGN KEY (category_id) REFERENCES product_categories(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. 收货地址（外卖用）
CREATE TABLE addresses (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id       BIGINT UNSIGNED NOT NULL,
  receiver_name VARCHAR(100) NOT NULL,
  phone         VARCHAR(30) NOT NULL,
  province      VARCHAR(100),
  city          VARCHAR(100),
  district      VARCHAR(100),
  street        VARCHAR(255),
  is_default    TINYINT(1) NOT NULL DEFAULT 0,
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_addr_user
    FOREIGN KEY (user_id) REFERENCES users(id),
  INDEX idx_addresses_user (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. 订单主表
CREATE TABLE orders (
  id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  order_no        VARCHAR(50) NOT NULL UNIQUE,
  user_id         BIGINT UNSIGNED NOT NULL,
  store_id        BIGINT UNSIGNED NOT NULL,
  delivery_mode   ENUM('self','delivery') NOT NULL,
  status          ENUM('pending_pay','paid','picked','done','cancelled') NOT NULL,
  total_amount    DECIMAL(10,2) NOT NULL,
  discount_amount DECIMAL(10,2) NOT NULL DEFAULT 0,
  payable_amount  DECIMAL(10,2) NOT NULL,
  paid_amount     DECIMAL(10,2) NOT NULL DEFAULT 0,
  payment_method  VARCHAR(50),
  created_at      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  paid_at         DATETIME NULL,
  finished_at     DATETIME NULL,
  cancel_reason   VARCHAR(255),
  CONSTRAINT fk_order_user  FOREIGN KEY (user_id)  REFERENCES users(id),
  CONSTRAINT fk_order_store FOREIGN KEY (store_id) REFERENCES stores(id),
  INDEX idx_orders_user        (user_id),
  INDEX idx_orders_store       (store_id),
  INDEX idx_orders_created_at  (created_at),
  INDEX idx_orders_status      (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 6. 订单明细（用于关联规则）
CREATE TABLE order_items (
  id                 BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  order_id           BIGINT UNSIGNED NOT NULL,
  product_id         BIGINT UNSIGNED NOT NULL,
  product_variant_id BIGINT UNSIGNED NULL,   -- 当前脚本不使用，为将来预留
  product_name_snap  VARCHAR(200) NOT NULL,
  unit_price         DECIMAL(10,2) NOT NULL,
  qty                INT NOT NULL,
  line_total         DECIMAL(10,2) NOT NULL,
  CONSTRAINT fk_item_order   FOREIGN KEY (order_id)   REFERENCES orders(id),
  CONSTRAINT fk_item_product FOREIGN KEY (product_id) REFERENCES products(id),
  INDEX idx_order_items_order   (order_id),
  INDEX idx_order_items_product (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 7. 外卖配送信息（马来西亚英文地址）
CREATE TABLE order_delivery_info (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  order_id      BIGINT UNSIGNED NOT NULL,
  receiver_name VARCHAR(100) NOT NULL,
  phone         VARCHAR(30) NOT NULL,
  full_address  VARCHAR(255) NOT NULL,
  lat           DECIMAL(10,7),
  lng           DECIMAL(10,7),
  remark        VARCHAR(255),
  CONSTRAINT fk_order_delivery
    FOREIGN KEY (order_id) REFERENCES orders(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 8. 可选：用户事件日志
CREATE TABLE user_events (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id     BIGINT UNSIGNED,
  event_type  VARCHAR(50) NOT NULL,
  event_time  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  product_id  BIGINT UNSIGNED NULL,
  order_id    BIGINT UNSIGNED NULL,
  store_id    BIGINT UNSIGNED NULL,
  meta        JSON NULL,
  INDEX idx_events_user      (user_id),
  INDEX idx_events_type_time (event_type, event_time),
  CONSTRAINT fk_event_user  FOREIGN KEY (user_id)  REFERENCES users(id),
  CONSTRAINT fk_event_prod  FOREIGN KEY (product_id) REFERENCES products(id),
  CONSTRAINT fk_event_order FOREIGN KEY (order_id)  REFERENCES orders(id),
  CONSTRAINT fk_event_store FOREIGN KEY (store_id) REFERENCES stores(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
